/* ============ 실적 상세 테이블 ============ */
(function(){
  const t = document.getElementById('perfTable');
  const monthHead = MONTHS.map(m=>`<th colspan="2">${m}</th>`).join('');
  const subHead = MONTHS.map(()=>'<th>건수</th><th>현장</th>').join('');
  let head = `<tr><th rowspan="2">구분</th><th rowspan="2">계획</th><th rowspan="2">대상<br>현장</th>${monthHead}<th rowspan="2">누계<br>건수</th><th rowspan="2">누계<br>현장</th><th rowspan="2">이행율</th></tr><tr>${subHead}</tr>`;
  let body = '';
  performance.forEach(p=>{
    const cells = p.monthly.map(m=>`<td>${m[0]}</td><td>${m[1]}</td>`).join('');
    body += `<tr><td>${p.label}</td><td>${p.plan}</td><td>${p.targetSites}</td>${cells}<td><b>${p.totalCount}</b></td><td>${p.totalSites}</td><td class="${p.rate<0.5?'hot':''}">${(p.rate*100).toFixed(1)}%</td></tr>`;
  });
  const sumPlan = performance.reduce((s,p)=>s+p.plan,0);
  const sumTarget = performance.reduce((s,p)=>s+p.targetSites,0);
  const sumMonthly = MONTHS.map((_,i)=>{
    const cnt = performance.reduce((s,p)=>s+p.monthly[i][0],0);
    const site = performance.reduce((s,p)=>s+p.monthly[i][1],0);
    return [cnt,site];
  });
  const sumCount = performance.reduce((s,p)=>s+p.totalCount,0);
  const sumSites = performance.reduce((s,p)=>s+p.totalSites,0);
  const sumRate = sumCount/sumPlan;
  const sumCells = sumMonthly.map(m=>`<td>${m[0]}</td><td>${m[1]}</td>`).join('');
  body += `<tr class="total"><td>건축+인프라</td><td>${sumPlan}</td><td>${sumTarget}</td>${sumCells}<td>${sumCount}</td><td>${sumSites}</td><td>${(sumRate*100).toFixed(1)}%</td></tr>`;
  t.innerHTML = head+body;
})();
